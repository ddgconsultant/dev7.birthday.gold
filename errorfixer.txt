  5. Safety Mechanisms

  1. Whitelist of Fixable Patterns
    - Undefined variable → Add isset() check
    - Null pointer → Add null coalesce
    - Typo in variable name (high confidence)
    - Missing array key → Add array_key_exists()
  2. Blacklist of No-Touch Areas
    - /core/classes/class.*.php (manual review only)
    - Database schema files
    - Authentication/security files
    - Payment processing
  3. Rollback System
    - Store original code before fix
    - Git commit each fix separately
    - Keep fix database in sync with git
  4. Testing Integration
    - Run php -l on fixed file (syntax check)
    - If file has tests, run them
    - Monitor error log for 24h post-fix

  CREATE TABLE IF NOT EXISTS bg_auto_error_fixes (
      fix_id INT AUTO_INCREMENT PRIMARY KEY,

      -- Error identification
      error_hash VARCHAR(64) UNIQUE,  -- MD5(file:line:type:message)
      error_file VARCHAR(500),
      error_line INT,
      error_message TEXT,
      error_type VARCHAR(50),         -- Fatal error, Warning, Notice, etc.
      error_context TEXT,             -- 20 lines of code around error

      -- Occurrence tracking
      first_seen DATETIME,
      last_seen DATETIME,
      occurrence_count INT DEFAULT 1,

      -- Fix workflow
      fix_status ENUM(
          'pending_analysis',          -- Just found, not analyzed yet
          'pending_review',            -- AI analyzed, waiting for approval
          'approved_pending_apply',    -- Richard approved, will apply next run
          'applied',                   -- Fix has been applied
          'rejected',                  -- Richard rejected the fix
          'failed_to_apply',          -- Application failed
          'auto_ignored'              -- Pattern ignored (blacklist)
      ) DEFAULT 'pending_analysis',

      -- AI analysis
      ai_analyzed_dt DATETIME NULL,
      ai_model VARCHAR(100),          -- e.g., 'claude-3-5-sonnet-20241022'
      ai_confidence INT,              -- 0-100
      ai_fix_type VARCHAR(50),        -- null_check, typo, missing_var, etc.
      ai_explanation TEXT,
      ai_review_reason TEXT,          -- Why it needs review (if applicable)

      -- Code changes
      original_code TEXT,             -- Backup for rollback
      proposed_fix TEXT,              -- Full fixed code block
      line_start INT,                 -- Where fix starts
      line_end INT,                   -- Where fix ends

      -- Application tracking
      applied_dt DATETIME NULL,
      applied_by VARCHAR(50),         -- 'auto_scheduler'
      git_commit_hash VARCHAR(40),    -- Git SHA after applying
      syntax_check_passed BOOLEAN,

      -- Review tracking
      reviewed_by INT NULL,           -- user_id from bg_users
      reviewed_dt DATETIME NULL,
      review_notes TEXT,
      review_token VARCHAR(64),       -- Unique token for review link

      -- Metadata
      created_dt DATETIME DEFAULT CURRENT_TIMESTAMP,
      updated_dt DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

      INDEX idx_status (fix_status),
      INDEX idx_error_hash (error_hash),
      INDEX idx_review_token (review_token),
      INDEX idx_first_seen (first_seen),
      INDEX idx_last_seen (last_seen)
  ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
